      \begin{itemize}

\begin{frame}{Problem Statement}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \tagline{Limitations in ordinary modelling}
      \begin{itemize}
        \item Ordinary models usually depend only on the present state.
        \item HIV-related behaviour may depend on past awareness, stigma, testing history, and adherence.
        \item Social behaviour is often simplified into fixed rates.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \tagline{Need addressed by this project}
      \begin{itemize}
        \item Include memory using a Caputo fractional derivative.
        \item Represent behavioural interventions through effective rates.
        \item Use numerical simulation to compare scenarios clearly.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.42cm}
  \keybox{SoftGold}{\textbf{Research focus:} How do fractional memory and social behaviour interventions affect simulated HIV dynamics in a SITA framework?}
\end{frame}

\begin{frame}{Fractional-Order Model}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.60\textwidth}
      \tagline{Caputo memory form}
      \[
        \CaputoD{t}{q}X(t)=F(t,X(t)),\qquad 0<q\leq1
      \]
      \[
      \begin{aligned}
        \CaputoD{t}{q}S &= \Lambda-\beff S(I+\eta T)-\mu S,\\
        \CaputoD{t}{q}I &= \beff S(I+\eta T)-(\taueff+\delta+\mu)I,\\
        \CaputoD{t}{q}T &= \taueff I-(\rhoeff+\mu)T,\\
        \CaputoD{t}{q}A &= \delta I+\rhoeff T-(d+\mu)A.
      \end{aligned}
      \]
    \end{column}
    \begin{column}{0.36\textwidth}
      \tagline{Intervention-adjusted rates}
      \[
        \beff=\beta_0(1-u_1)(1-u_2)
      \]
      \[
        \taueff=\tau(1+u_3)
      \]
      \[
        \rhoeff=\rho(1-u_4)
      \]
      \keybox{SoftTeal}{When $q=1$, the model is ordinary. When $q<1$, past states influence current change.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Numerical Scheme + Implementation}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \tagline{Numerical idea}
      \begin{itemize}
        \item Time interval is divided into discrete steps.
        \item A predictor estimates the next state.
        \item A corrector updates the state using fractional-memory weights.
        \item One engine supports baseline, memory, intervention, and sensitivity experiments.
      \end{itemize}
    \end{column}
