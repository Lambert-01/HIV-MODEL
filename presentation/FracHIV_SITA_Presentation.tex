\begin{frame}{Background / Motivation}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \tagline{Rwanda-aware public-health motivation}
      \begin{itemize}
        \item HIV transmission is shaped by biology, treatment access, and social behaviour.
        \item Awareness, safer behaviour, testing, treatment-seeking, and adherence can change long-term outcomes.
        \item In Rwanda, continued prevention and adherence support remain important alongside treatment progress.
      \end{itemize}
      \vspace{0.18cm}
      \keybox{SoftTeal}{\textbf{Why mathematics?} A model gives a controlled way to compare intervention strategies and understand possible long-term behaviour.}
    \end{column}
    \begin{column}{0.37\textwidth}
      \compactcard{Blue}{Context}{HIV in Rwanda}\\[0.16cm]
      \compactcard{Teal}{Mechanism}{Social behaviour}\\[0.16cm]
      \compactcard{Gold}{Method}{Fractional memory}\\[0.16cm]
      \compactcard{Red}{Output}{Simulation evidence}
    \end{column}
  \end{columns}
\end{frame}

% 3
\begin{frame}{Problem Statement + Literature Gap}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \tagline{Limitations in ordinary modelling}
      \begin{itemize}
        \item Ordinary models usually depend only on the present state.
        \item HIV-related behaviour may depend on past awareness, stigma, testing history, and adherence.
        \item Social behaviour is often simplified into fixed rates.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \tagline{Literature gap}
      \begin{itemize}
        \item Classical HIV models use ODE compartments such as SICA/SITA.
        \item Fractional models use Caputo derivatives for memory effects.
        \item Few works combine fractional HIV dynamics, behaviour controls, and an interactive dashboard.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.34cm}
  \keybox{SoftGold}{\textbf{Research focus:} a fractional-order SITA model that connects memory effects, social behaviour interventions, numerical simulation, and dashboard exploration.}
\end{frame}

\begin{frame}{Fractional-Order SITA Model}
  \scriptsize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \tagline{Objective 1: model formulation}
      \[
        \CaputoD{t}{q}X(t)=F(t,X(t)),\qquad 0<q\leq1
      \]
      \[
      \begin{aligned}
        \CaputoD{t}{q}S &= \Lambda-\lambda(t)S-\mu S,\\
        \CaputoD{t}{q}I &= \lambda(t)S-(\taueff+\delta+\mu)I,\\
        \CaputoD{t}{q}T &= \taueff I-(\rhoeff+\mu)T,\\
        \CaputoD{t}{q}A &= \delta I+\rhoeff T-(d+\mu)A.
      \end{aligned}
      \]
      \vspace{-0.10cm}
      \[
      \begin{aligned}
        \lambda(t)&=\frac{\beff(I+\eta T)}{N},\\[-1pt]
        \beff&=\beta_0(1-u_1)(1-u_2),\quad
        \taueff=\tau(1+u_3),\\[-1pt]
        \rhoeff&=\rho(1-u_4).
      \end{aligned}
      \]
    \end{column}
    \begin{column}{0.39\textwidth}
      \tagline{Meaning of interventions}
      \keybox{SoftGold}{$u_1,u_2$ reduce effective transmission through awareness and safer behaviour.}
      \vspace{0.12cm}
      \keybox{SoftTeal}{$u_3$ increases testing/treatment movement from infected to treated.}
      \vspace{0.12cm}
      \keybox{SoftBlue}{$u_4$ reduces progression from treatment to AIDS stage through adherence support.}
      \vspace{0.12cm}
      \keybox{SoftGold}{When $q=1$ the model becomes ordinary; when $q<1$ previous states influence the present trajectory.}
    \end{column}
  \end{columns}
\end{frame}

% 7
\begin{frame}{Objective 2: Mathematical Analysis Evidence}
  \scriptsize
  \vspace{-0.06cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \tagline{Positivity proof}
      \[
      \begin{array}{ll}
      S=0:\; \CaputoD{t}{q}S=\Lambda\ge0, &
      I=0:\; \CaputoD{t}{q}I=\lambda(t)S\ge0,\\[2pt]
      T=0:\; \CaputoD{t}{q}T=\taueff I\ge0, &
      A=0:\; \CaputoD{t}{q}A=\delta I+\rhoeff T\ge0.
      \end{array}
      \]
      \keybox{SoftBlue}{Therefore the vector field points inward on the boundary, so the nonnegative region $\mathbb{R}_+^4$ is positively invariant.}
      \vspace{0.14cm}
      \tagline{Boundedness proof}
      \[
        N=S+I+T+A,\qquad
        \CaputoD{t}{q}N=\Lambda-\mu N-dA\le \Lambda-\mu N.
      \]
      \[
        \limsup_{t\to\infty}N(t)\le \frac{\Lambda}{\mu}.
      \]
    \end{column}
    \begin{column}{0.47\textwidth}
      \tagline{Disease-free equilibrium and threshold}
      \[
        E_0=\left(\frac{\Lambda}{\mu},0,0,0\right),\qquad
        \Rzero=\rho(FV^{-1}).
      \]
      \keybox{SoftGold}{$F$ contains new-infection terms and $V$ contains transition/removal terms of the infected subsystem. Thus $\Rzero$ is obtained by the next-generation matrix method.}
      \vspace{0.14cm}
      \tagline{Fractional stability condition}
      \[
        |\arg(\lambda_i)|>\frac{q\pi}{2}\quad\text{for every eigenvalue }\lambda_i.
      \]
      \keybox{SoftTeal}{When $\Rzero<1$ and the eigenvalues satisfy this Matignon condition, the disease-free equilibrium is locally asymptotically stable.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Numerical Scheme + Implementation}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \tagline{Numerical idea}
      \begin{itemize}
        \item Time interval is divided into discrete steps.
        \item A predictor estimates the next state.
        \item A corrector updates the state using fractional-memory weights.
        \item One engine supports baseline, memory, intervention, and sensitivity experiments.
